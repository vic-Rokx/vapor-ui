Installation

// In main.zig — build CSS keyframes once at startup const Marquee = @import("components/Marquee.zig"); Marquee.new();

Usage

const Marquee = @import("components/Marquee.zig"); const items = [_]Marquee.MarqueeItem{ .{ .title = "Slide 1", .subtitle = "desc", .color = .hex("#3b82f6") }, .{ .title = "Slide 2", .subtitle = "desc", .color = .hex("#8b5cf6") }, // ... }; // Horizontal scroll left at normal speed Marquee.render(.{ .items = &items, .direction = .left, .speed_ms = 10000, }); // Vertical scroll up, fast Marquee.render(.{ .items = &items, .direction = .up, .speed_ms = 5000, });

API Reference

MarqueeItem

pub const MarqueeItem = struct { title: []const u8, subtitle: []const u8 = "", // optional label below title color: Color = .palette(.tint), // background color of slide };

Custom item renderer via create() + Component()

fn mySlide(item: Marquee.MarqueeItem, extra: *MyCtx) void { Stack() .background(item.color) .children({ Text(item.title).end(); // ...anything else using `extra` }); } _ = Marquee.create(.{ .items = &items, .direction = .left, }).Component(mySlide, .{ &my_ctx });

Marquee fields

items: []const MarqueeItem, // slide data direction: Direction = .left, // .left | .right | .up | .down speed_ms: u32 = 8000, // duration per cycle — lower = faster item_width: f32 = 240, // slide width (px) item_height: f32 = 160, // slide height (px) gap: f32 = 16, // gap between slides (px) visible_count: f32 = 3, // slides visible in viewport
